    def _update_recent_files_menu(self):
        """Update recent files menu with current list"""
        # Clear existing actions
        self.recent_files_menu.clear()
        
        # Get recent files from settings
        recent_files = self.settings.value("recent_files", [])
        if not isinstance(recent_files, list):
            recent_files = []
        
        # Add actions for each recent file
        if recent_files:
            for file_path in recent_files[:self.max_recent_files]:
                if Path(file_path).exists():
                    # Create action with filename and path
                    filename = Path(file_path).name
                    action = QAction(f"{filename}", self)
                    action.setStatusTip(str(file_path))
                    action.setData(file_path)
                    action.triggered.connect(lambda checked, p=file_path: self._open_recent_file(p))
                    self.recent_files_menu.addAction(action)
            
            # Add separator and clear action
            self.recent_files_menu.addSeparator()
            clear_action = QAction("Clear Recent Files", self)
            clear_action.triggered.connect(self._clear_recent_files)
            self.recent_files_menu.addAction(clear_action)
        else:
            # No recent files
            no_files_action = QAction("(No recent files)", self)
            no_files_action.setEnabled(False)
            self.recent_files_menu.addAction(no_files_action)
    
    def _add_to_recent_files(self, file_path: str):
        """Add file to recent files list"""
        recent_files = self.settings.value("recent_files", [])
        if not isinstance(recent_files, list):
            recent_files = []
        
        # Remove if already in list
        if file_path in recent_files:
            recent_files.remove(file_path)
        
        # Add to front
        recent_files.insert(0, file_path)
        
        # Keep only max_recent_files
        recent_files = recent_files[:self.max_recent_files]
        
        # Save to settings
        self.settings.setValue("recent_files", recent_files)
        
        # Update menu
        self._update_recent_files_menu()
    
    def _open_recent_file(self, file_path: str):
        """Open a file from recent files list"""
        if not Path(file_path).exists():
            QMessageBox.warning(
                self,
                "File Not Found",
                f"The file no longer exists:\n{file_path}"
            )
            # Remove from recent files
            recent_files = self.settings.value("recent_files", [])
            if file_path in recent_files:
                recent_files.remove(file_path)
                self.settings.setValue("recent_files", recent_files)
                self._update_recent_files_menu()
            return
        
        # Determine file type and open accordingly
        file_ext = Path(file_path).suffix.lower()
        
        if file_ext == '.dpa':
            # Project file
            self._open_project_file(file_path)
        elif file_ext in ['.arxml', '.xml']:
            # Could be DEF or configuration file - try DEF first
            try:
                self._open_def_file_path(file_path)
            except:
                # If DEF fails, might be a configuration file
                QMessageBox.information(
                    self,
                    "Open File",
                    "Please use 'Open DEF File' or 'Open Project' for this file."
                )
        else:
            QMessageBox.warning(
                self,
                "Unknown File Type",
                f"Cannot open file with extension: {file_ext}"
            )
    
    def _open_project_file(self, file_path: str):
        """Helper to open project file"""
        try:
            self.statusbar.showMessage("Loading project...")
            self.current_project, failed_modules = self.workspace_manager.load_project(Path(file_path))
            self.current_project_file = Path(file_path)
            self.tree_view.set_project(self.current_project)
            
            # Enable project actions
            self.save_project_action.setEnabled(True)
            self.project_properties_action.setEnabled(True)
            self.add_module_action.setEnabled(True)
            
            # Update mode label
            self.mode_label.setText(f"Mode: Project ({self.current_project.name})")
            
            status_msg = f"Loaded project: {self.current_project.name}"
            
            if failed_modules:
                error_details = "\n".join([f"• {name}: {error}" for name, error in failed_modules])
                QMessageBox.warning(
                    self,
                    "Project Loaded with Errors",
                    f"Project loaded, but {len(failed_modules)} module(s) failed:\n\n{error_details}\n\n"
                    f"Successfully loaded: {len(self.current_project.module_managers)} module(s)"
                )
                status_msg += f" ({len(failed_modules)} errors)"
            
            self.statusbar.showMessage(status_msg, 5000)
            
        except Exception as e:
            QMessageBox.critical(self, "Error", f"Failed to load project:\n{e}")
    
    def _open_def_file_path(self, file_path: str):
        """Helper to open DEF file"""
        self.statusbar.showMessage("Loading DEF file...")
        QApplication.setOverrideCursor(Qt.WaitCursor)
        
        try:
            # Parse DEF file
            self.module_def = self.def_parser.parse_module_def_file(Path(file_path))
            
            # Create new configuration manager
            self.config_manager = ConfigurationManager(self.module_def)
            
            # Update tree
            self.tree_view.set_module_def(self.module_def, self.config_manager)
            
            # Update mode label
            self.mode_label.setText("Mode: Single Module")
            
            # Enable actions
            self.new_config_action.setEnabled(True)
            self.open_value_action.setEnabled(True)
            self.generate_action.setEnabled(True)
            
            self.statusbar.showMessage(f"Loaded: {self.module_def.short_name}", 3000)
            
        except Exception as e:
            QMessageBox.critical(self, "Error", f"Failed to load DEF file:\n{str(e)}")
        finally:
            QApplication.restoreOverrideCursor()
    
    def _clear_recent_files(self):
        """Clear all recent files"""
        reply = QMessageBox.question(
            self,
            "Clear Recent Files",
            "Clear all recent files from the list?",
            QMessageBox.Yes | QMessageBox.No,
            QMessageBox.No
        )
        
        if reply == QMessageBox.Yes:
            self.settings.setValue("recent_files", [])
            self._update_recent_files_menu()
